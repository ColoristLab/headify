#!/usr/bin/env node

import cac from "cac";

import pkg from "../package.json" with { type: "json" };
import { logger } from "./logger.ts";

const cli = cac(pkg.name);

cli.help().version(pkg.version);

cli
  .command("<...files>", "Prepend banner headers to files.", {
    ignoreOptionDefaultValue: true,
    allowUnknownOptions: true,
  })
  .option("-t, --template <path>", "Path to Eta template file")
  .option("-e, --exclude <...patterns>", "Exclude files or directories (supports glob)", {
    type: [],
  })
  .option("--dry-run", "Preview changes without writing files")
  .action(async (files: string[], options) => {
    const [{ headify }, { reportResult }] = await Promise.all([
      import("./headify.ts"),
      import("./reporter.ts"),
    ]);

    const result = await headify(files, options);

    reportResult(pkg.version, result);
  });

/**
 * 解析命令行参数并执行匹配的命令。
 */
export async function runCli(argv: string[] = process.argv): Promise<void> {
  try {
    cli.parse(argv, { run: false });

    await cli.runMatchedCommand();
  } catch (error: unknown) {
    logger.error(error);
    process.exitCode = 1;
  }
}

void runCli();

import { basename, extname, relative, resolve, sep } from "node:path";

import type { PackageJson } from "pkg-types";

import type { FileInfo, TemplateContext } from "./types.js";

/**
 * 构造当前文件的模板上下文。
 */
export function createContext(
  packageMetadata: PackageJson,
  filePath: string,
  cwd: string,
): TemplateContext {
  const absolutePath = resolve(cwd, filePath);
  const relativePath = relative(cwd, absolutePath);
  const fileName = basename(absolutePath);
  const extension = extname(fileName);

  const file: FileInfo = {
    path: relativePath.split(sep).join("/"),
    name: fileName,
    basename: fileName.slice(0, fileName.length - extension.length),
    ext: extension,
  };

  return {
    package: packageMetadata,
    year: new Date().getFullYear(),
    file,
  };
}


src/file.ts

import { readFile, stat, writeFile } from "node:fs/promises";
import { glob } from "node:fs/promises";
import { resolve } from "node:path";

import type { HeadifyOptions } from "./types.js";

type FindFilesOptions = Required<Pick<HeadifyOptions, "exclude" | "cwd">>;

/**
 * 查找目标文件，只返回普通文件。
 */
export async function findFiles(
  patterns: string | string[],
  options: FindFilesOptions,
): Promise<string[]> {
  const { cwd, exclude } = options;
  const files = new Set<string>();

  for await (const file of glob(patterns, {
    cwd,
    exclude,
  })) {
    const absolutePath = resolve(cwd, file);
    const fileStat = await stat(absolutePath);

    if (fileStat.isFile()) {
      files.add(absolutePath);
    }
  }

  return [...files];
}

/**
 * 规范 banner 的尾部空白。
 */
function normalizeBanner(banner: string): string {
  return banner.trimEnd();
}

/**
 * 判断文件内容是否已经包含指定 banner。
 */
function hasBannerContent(content: string, banner: string): boolean {
  const normalizedBanner = normalizeBanner(banner);

  return (
    content.startsWith(normalizedBanner) &&
    (content.length === normalizedBanner.length ||
      content.startsWith(`${normalizedBanner}\n`) ||
      content.startsWith(`${normalizedBanner}\r\n`))
  );
}

/**
 * 检查文件是否已经包含当前 banner。
 */
export async function hasBanner(file: string, banner: string): Promise<boolean> {
  const content = await readFile(file, "utf8");

  return hasBannerContent(content, banner);
}

/**
 * 在文件开头插入 banner。
 *
 * 返回 false 表示文件已经存在相同的 banner。
 */
export async function prependBanner(file: string, banner: string): Promise<boolean> {
  const content = await readFile(file, "utf8");
  const normalizedBanner = normalizeBanner(banner);

  if (hasBannerContent(content, normalizedBanner)) {
    return false;
  }

  const separator = content.length > 0 ? "\n" : "";

  await writeFile(file, `${normalizedBanner}${separator}${content}`, "utf8");

  return true;
}

import { readFile } from "node:fs/promises";
import { relative, resolve } from "node:path";

import type { PackageJson } from "pkg-types";

import { createContext } from "./context.js";
import { findFiles, hasBanner, prependBanner } from "./file.js";
import { renderBanner, resolveTemplate } from "./renderer.js";
import type { FileStatus, FileResult, HeadifyOptions, HeadifyResult } from "./types.js";

export const defaultOptions: Required<HeadifyOptions> = {
  cwd: process.cwd(),
  template: "",
  exclude: [],
  dryRun: false,
};

export async function headify(
  patterns: string[],
  options: HeadifyOptions = {},
): Promise<HeadifyResult> {
  const config = {
    ...defaultOptions,
    ...options,
  };

  validateOptions(patterns, config);

  const { cwd, template, exclude, dryRun } = config;
  const startedAt = Date.now();

  const files = await findFiles(patterns, { cwd, exclude });

  // 没有匹配文件时直接返回空结果，不读取 package.json。
  if (files.length === 0) {
    return {
      template,
      dryRun,
      total: 0,
      updated: 0,
      unchanged: 0,
      wouldUpdate: 0,
      elapsed: Date.now() - startedAt,
      files: [],
    };
  }

  const packagePath = resolve(cwd, "package.json");
  const packageMetadata = JSON.parse(await readFile(packagePath, "utf8")) as PackageJson;

  // 整个任务只加载一次模板。
  const templateMethod = await resolveTemplate(template);

  const results: FileResult[] = [];

  let updated = 0;
  let unchanged = 0;
  let wouldUpdate = 0;

  for (const file of files) {
    const context = createContext(packageMetadata, file, cwd);
    const path = relative(cwd, file).replaceAll("\\", "/");
    const banner = await renderBanner(templateMethod, context);

    let status: FileStatus;

    if (dryRun) {
      if (await hasBanner(file, banner)) {
        unchanged++;
        status = "unchanged";
      } else {
        wouldUpdate++;
        status = "wouldUpdate";
      }
    } else if (await prependBanner(file, banner)) {
      updated++;
      status = "updated";
    } else {
      unchanged++;
      status = "unchanged";
    }

    results.push({ path, status });
  }

  return {
    template,
    dryRun,
    total: results.length,
    updated,
    unchanged,
    wouldUpdate,
    elapsed: Date.now() - startedAt,
    files: results,
  };
}

/**
 * 验证 headify 参数。
 */
function validateOptions(patterns: unknown, options: Required<HeadifyOptions>): void {
  if (!Array.isArray(patterns) || !patterns.every((pattern) => typeof pattern === "string")) {
    throw new TypeError("Expected patterns to be an array of strings.");
  }

  if (typeof options.cwd !== "string") {
    throw new TypeError('Expected "cwd" to be a string.');
  }

  if (typeof options.template !== "string") {
    throw new TypeError('Expected "template" to be a string.');
  }

  if (
    !Array.isArray(options.exclude) ||
    !options.exclude.every((pattern) => typeof pattern === "string")
  ) {
    throw new TypeError('Expected "exclude" to be an array of strings.');
  }

  if (typeof options.dryRun !== "boolean") {
    throw new TypeError('Expected "dryRun" to be a boolean.');
  }
}

src/load-template.ts

import { createRequire } from "node:module";
import { extname, resolve } from "node:path";
import { pathToFileURL } from "node:url";

import type { BannerTemplate } from "./types.ts";

const require = createRequire(import.meta.url);

const cjsExtensions = new Set([".cjs", ".cts"]);

export default async function loadTemplate(templatePath: string): Promise<BannerTemplate> {
  const filePath = resolve(templatePath);
  const extension = extname(filePath).toLowerCase();

  let template: unknown;

  if (cjsExtensions.has(extension)) {
    template = require(filePath);
  } else if ([".js", ".mjs", ".ts", ".mts"].includes(extension)) {
    const mod = await import(pathToFileURL(filePath).href);
    template = mod.default;
  } else {
    throw new TypeError(`Unsupported template extension: ${extension}`);
  }

  if (typeof template !== "function") {
    throw new TypeError(`Template "${filePath}" must export a function as its default export.`);
  }

  return template as BannerTemplate;
}


src/logger.ts

import { styleText } from "node:util";

type LogMethod = (...messages: unknown[]) => void;
type LogColor = Parameters<typeof styleText>[0];
type InspectColor = Exclude<LogColor, readonly unknown[]>;

const colors = {
  info: "cyan",
  success: "green",
  stack: "gray",
  location: "cyan",
} as const satisfies Record<string, InspectColor>;

/**
 * 格式化带背景色的标签。
 */
function formatLabel(label: string, background: InspectColor): string {
  return styleText(["black", "bold", background], ` ${label} `);
}

/**
 * 将未知类型的值转换为字符串。
 */
function stringify(value: unknown): string {
  if (value instanceof Error) {
    return value.message;
  }

  if (typeof value === "string") {
    return value;
  }

  try {
    return String(value);
  } catch {
    return "[Unprintable value]";
  }
}

/**
 * 移除文件 URL 的协议前缀。
 *
 * 例如：
 * file:///C:/project/src/index.ts:10:20
 * -> C:/project/src/index.ts:10:20
 */
function normalizeStackLocation(location: string): string {
  return location.replace(/^file:\/\//, "");
}

/**
 * 格式化错误堆栈行。
 *
 * - 只有 `at` 使用灰色。
 * - 函数名和其他普通文本使用终端默认颜色。
 * - 文件路径及行列号使用青色。
 * - 移除 file:/// 前缀。
 * - 原始缩进减少两个空格。
 */
function formatStackLine(line: string): string {
  const trimmed = line.trimStart();
  const originalIndent = line.slice(0, line.length - trimmed.length);

  // 缩进
  const indent = originalIndent.slice(0, Math.max(0, originalIndent.length - 0));

  const frame = trimmed.match(/^(at)(\s+)(.*)$/);

  if (!frame) {
    return indent + styleText(colors.stack, trimmed);
  }

  const [, prefix, separator, content] = frame;

  if (prefix === undefined || separator === undefined || content === undefined) {
    return indent + styleText(colors.stack, trimmed);
  }

  // at functionName (file:///path/file.mjs:10:20)
  const parenthesized = content.match(/^(.*?)\s+\((.+)\)$/);

  if (parenthesized) {
    const [, method, rawLocation] = parenthesized;

    if (method !== undefined && rawLocation !== undefined) {
      const location = normalizeStackLocation(rawLocation);

      return (
        indent +
        styleText(colors.stack, prefix) +
        separator +
        method +
        " (" +
        styleText(colors.location, location) +
        ")"
      );
    }
  }

  // at file:///path/file.mjs:10:20
  // at async node:internal/modules/esm/loader:646:26
  const plainLocation = content.match(/^(.*?)(:\d+(?::\d+)?)$/);

  if (plainLocation) {
    const [, rawFile, position] = plainLocation;

    if (rawFile !== undefined && position !== undefined) {
      const file = normalizeStackLocation(rawFile);

      // async 等前缀保持默认颜色，仅路径和行列号使用青色。
      const asyncPrefix = file.match(/^(async\s+)(.+)$/);

      if (asyncPrefix) {
        const [, asyncText, actualFile] = asyncPrefix;

        if (asyncText !== undefined && actualFile !== undefined) {
          return (
            indent +
            styleText(colors.stack, prefix) +
            separator +
            asyncText +
            styleText(colors.location, actualFile + position)
          );
        }
      }

      return (
        indent +
        styleText(colors.stack, prefix) +
        separator +
        styleText(colors.location, file + position)
      );
    }
  }

  // 无法识别的堆栈格式：只将 at 设置为灰色。
  return indent + styleText(colors.stack, prefix) + separator + content;
}

/**
 * 格式化错误堆栈，支持递归处理 Error.cause。
 */
function formatError(error: Error, level = 0): string[] {
  const lines: string[] = [];
  const indent = "  ".repeat(level);

  if (level > 0) {
    lines.push(`${indent}${styleText("yellow", "Caused by:")} ${error.name}: ${error.message}`);
  }

  if (error.stack) {
    const stackLines = error.stack.split("\n").slice(1);

    for (const line of stackLines) {
      lines.push(`${indent}${formatStackLine(line)}`);
    }
  }

  if (error.cause instanceof Error) {
    if (lines.length > 0) {
      lines.push("");
    }

    lines.push(...formatError(error.cause, level + 1));
  }

  return lines;
}

/**
 * 输出上下各带一行空行的日志。
 */
function writeSpaced(callback: () => void): void {
  console.log();
  callback();
  console.log();
}

/**
 * 创建普通日志方法。
 */
function write(label: string, color: InspectColor): LogMethod {
  return (...messages) => {
    console.log(`${styleText(color, label)} ${messages.map(stringify).join(" ")}`);
  };
}

/**
 * 创建带背景色标签的警告日志方法。
 */
function writeBadge(label: string, background: InspectColor): LogMethod {
  return (...messages) => {
    writeSpaced(() => {
      console.warn(`${formatLabel(label, background)} ${messages.map(stringify).join(" ")}`);
    });
  };
}

/**
 * 创建错误日志方法。
 */
function writeError(): LogMethod {
  return (...messages) => {
    writeSpaced(() => {
      for (const message of messages) {
        if (!(message instanceof Error)) {
          console.error(`${formatLabel("ERROR", "bgRed")} ${stringify(message)}`);
          continue;
        }

        // 输出错误提示。
        console.error(`${formatLabel("ERROR", "bgRed")} ${message.message}`);

        // 格式化错误堆栈。
        const stack = formatError(message);

        if (stack.length > 0) {
          // 错误提示与堆栈之间保留一行空行。
          console.error();
          console.error(stack.join("\n"));
        }
      }
    });
  };
}

export const logger = {
  log: (...messages: unknown[]) => console.log(...messages),

  info: write("ℹ", colors.info),

  success: write("✔", colors.success),

  warn: writeBadge("WARN", "bgYellow"),

  error: writeError(),
};

import loadTemplate from "./load-template.ts";

import type { TemplateContext, BannerTemplate } from "./types.js";

/**
 * 加载模板函数。
 */
export async function resolveTemplate(template?: string): Promise<BannerTemplate> {
  return template ? loadTemplate(template) : defaultTemplate;
}

/**
 * 渲染文件头部内容。
 */
export async function renderBanner(
  template: BannerTemplate,
  context: TemplateContext,
): Promise<string> {
  const output = await template(context);

  if (typeof output !== "string") {
    throw new TypeError("Failed to render banner template.");
  }

  return output;
}


src/reporter.ts

import { styleText } from "node:util";

import pkg from "../package.json" with { type: "json" };
import { logger } from "./logger.js";
import type { FileStatus, HeadifyResult } from "./types.js";

const colors = {
  muted: "gray",
  info: "cyan",
  success: "green",
} as const;

/**
 * 输出单个文件的处理结果。
 */
function reportFile(path: string, status: FileStatus): void {
  const separator = path.lastIndexOf("/");
  const directory = separator === -1 ? "" : path.slice(0, separator + 1);
  const filename = path.slice(separator + 1);

  const suffix = {
    updated: "",
    unchanged: ` ${styleText(colors.muted, "(Unchanged)")}`,
    wouldUpdate: ` ${styleText(colors.info, "(Would update)")}`,
  }[status];

  logger.info(`${styleText(colors.muted, directory)}${filename}${suffix}`);
}

/**
 * 根据任务结果输出控制台报告。
 */
export function reportResult(cliVersion: string, result: HeadifyResult): void {
  if (result.total === 0) {
    logger.warn("No files matched the specified patterns.");
    return;
  }

  logger.info(`${pkg.name} v${cliVersion}`);
  logger.info(`Matched ${result.total} files`);
  logger.info(`Template: ${result.template || "default"}`);

  if (result.dryRun) {
    logger.info("Dry run: no files will be modified");
  }

  for (const file of result.files) {
    reportFile(file.path, file.status);
  }

  const summary = result.dryRun
    ? `Would update: ${result.wouldUpdate} Unchanged: ${result.unchanged}`
    : `Updated: ${result.updated} Unchanged: ${result.unchanged}`;

  logger.info(summary);
  logger.success(`Completed in ${styleText(colors.success, `${result.elapsed} ms`)}`);
}

import { mkdtemp, mkdir, readFile, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";

import { afterEach, describe, expect, it } from "vitest";

import { findFiles, hasBanner, prependBanner } from "../src/file.js";

let cwd: string;

async function setup() {
  cwd = await mkdtemp(join(tmpdir(), "headify-file-test-"));
  return cwd;
}

afterEach(async () => {
  if (cwd) await rm(cwd, { recursive: true, force: true });
});

describe("findFiles", () => {
  it("查找匹配的普通文件", async () => {
    const root = await setup();
    await mkdir(join(root, "src"), { recursive: true });
    await writeFile(join(root, "src", "a.ts"), "a");
    await writeFile(join(root, "src", "b.ts"), "b");

    const files = await findFiles(["src/*.ts"], { cwd: root, exclude: [] });

    expect(files).toHaveLength(2);
    expect(files).toEqual(

    );
  });

  it("支持多个 glob 匹配模式，且不会产生重复结果", async () => {
    const root = await setup();
    await writeFile(join(root, "a.ts"), "a");
    await writeFile(join(root, "b.ts"), "b");

    const files = await findFiles(["*.ts", "a.ts"], { cwd: root, exclude: [] });

    expect(files).toHaveLength(2);
  });

    const root = await setup();
    await mkdir(join(root, "dist"), { recursive: true });
    await writeFile(join(root, "src.ts"), "src");
    await writeFile(join(root, "dist", "bundle.js"), "dist");

    const files = await findFiles(["**/*"], {

test/helpers.ts

import { mkdtemp, readFile, rm, writeFile, mkdir } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join, dirname } from "node:path";

export async function createProject() {
  const cwd = await mkdtemp(join(tmpdir(), "headify-test-"));

  await writeFile(
    join(cwd, "package.json"),
    JSON.stringify(
      {
        name: "test-project",
        version: "1.2.3",
        homepage: "https://example.com",
        author: "Test Author",
        license: "MIT",
      },
      null,
      2,
    ),
  );

  return {
    cwd,
    async writeFile(relativePath: string, content: string) {
      const path = join(cwd, relativePath);
      await mkdir(dirname(path), { recursive: true });
      await writeFile(path, content, "utf8");
      return path;
    },
    readFile(relativePath: string) {
      return readFile(join(cwd, relativePath), "utf8");
    },
    async cleanup() {
      await rm(cwd, { recursive: true, force: true });
    },
  };
}

    expect(

        package: { name: "demo" },
        year: 2026,
