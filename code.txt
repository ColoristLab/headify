import { describe, expect, it } from "vitest";
import { createContext } from "../src/context.js";

describe("createContext", () => {
  const packageMetadata = {
    name: "demo",
    version: "1.0.0",
  };

  it("builds package, year and file context", () => {
    const before = new Date().getFullYear();
    const context = createContext(packageMetadata, "src/index.ts", "/project");

    expect(context.package).toBe(packageMetadata);
    expect(context.year).toBe(before);
    expect(context.file).toEqual({
      path: "src/index.ts",
      name: "index.ts",
      basename: "index",
      ext: ".ts",
    });
  });

  it("normalizes Windows separators in the relative path", () => {
    const context = createContext(packageMetadata, "src/components/button.js", "/project");

    expect(context.file.path).toBe("src/components/button.js");
  });

  it("supports files without extensions", () => {
    const context = createContext(packageMetadata, "LICENSE", "/project");

    expect(context.file).toEqual({
      path: "LICENSE",
      name: "LICENSE",
      basename: "LICENSE",
      ext: "",
    });
  });

  it("supports dotfiles", () => {
    const context = createContext(packageMetadata, ".editorconfig", "/project");

    expect(context.file).toEqual({
      path: ".editorconfig",
      name: ".editorconfig",
      basename: ".editorconfig",
      ext: "",
    });
  });

  it("resolves absolute file paths and keeps paths relative to cwd", () => {
    const context = createContext(packageMetadata, "/project/lib/index.js", "/project");

    expect(context.file.path).toBe("lib/index.js");
    expect(context.file.name).toBe("index.js");
  });
});


test/file.test.ts

import { mkdtemp, mkdir, readFile, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { afterEach, describe, expect, it } from "vitest";
import { findFiles, hasBanner, prependBanner } from "../src/file.js";

let cwd: string;

async function setup() {
  cwd = await mkdtemp(join(tmpdir(), "headmark-file-test-"));
  return cwd;
}

afterEach(async () => {
  if (cwd) await rm(cwd, { recursive: true, force: true });
});

describe("findFiles", () => {
  it("finds matching regular files", async () => {
    const root = await setup();
    await mkdir(join(root, "src"), { recursive: true });
    await writeFile(join(root, "src", "a.ts"), "a");
    await writeFile(join(root, "src", "b.ts"), "b");
    await writeFile(join(root, "src", "c.js"), "c");

    const files = await findFiles(["src/*.ts"], { cwd: root, exclude: [] });

    expect(files).toHaveLength(2);
    expect(files).toEqual(expect.arrayContaining([
      join(root, "src", "a.ts"),
      join(root, "src", "b.ts"),
    ]));
  });

  it("supports multiple glob patterns without duplicate results", async () => {
    const root = await setup();
    await writeFile(join(root, "a.ts"), "a");
    await writeFile(join(root, "b.ts"), "b");

    const files = await findFiles(["*.ts", "a.ts"], { cwd: root, exclude: [] });

    expect(files).toHaveLength(2);
  });

  it("honors exclude patterns", async () => {
    const root = await setup();
    await mkdir(join(root, "dist"), { recursive: true });
    await writeFile(join(root, "src.ts"), "src");
    await writeFile(join(root, "dist", "bundle.js"), "dist");

    const files = await findFiles(["**/*"], {
      cwd: root,
      exclude: ["dist/**", "package.json"],
    });

    expect(files).toEqual([join(root, "src.ts")]);
  });

  it("returns an empty array when nothing matches", async () => {
    const root = await setup();

    await expect(findFiles(["missing/**/*.ts"], { cwd: root, exclude: [] }))
      .resolves.toEqual([]);
  });
});

describe("hasBanner", () => {
  it("returns true when the banner is at the beginning of a file", async () => {
    const root = await setup();
    const file = join(root, "a.ts");
    await writeFile(file, "/* banner */\nconst value = 1;\n");

    await expect(hasBanner(file, "/* banner */")).resolves.toBe(true);
  });

  it("accepts CRLF after the banner", async () => {
    const root = await setup();
    const file = join(root, "a.ts");
    await writeFile(file, "/* banner */\r\nconst value = 1;\r\n");

    await expect(hasBanner(file, "/* banner */\n\n")).resolves.toBe(true);
  });

  it("returns false when the banner appears later in the file", async () => {
    const root = await setup();
    const file = join(root, "a.ts");
    await writeFile(file, "const value = 1;\n/* banner */");

    await expect(hasBanner(file, "/* banner */")).resolves.toBe(false);
  });

  it("does not treat a longer prefix as the same banner", async () => {
    const root = await setup();
    const file = join(root, "a.ts");
    await writeFile(file, "/* banner */ extra\ncontent");

    await expect(hasBanner(file, "/* banner */")).resolves.toBe(false);
  });
});

describe("prependBanner", () => {
  it("prepends the banner and preserves the original content", async () => {
    const root = await setup();
    const file = join(root, "a.ts");
    await writeFile(file, "const value = 1;\n");

    await expect(prependBanner(file, "/* banner */\n")).resolves.toBe(true);
    await expect(readFile(file, "utf8")).resolves.toBe("/* banner */\nconst value = 1;\n");
  });

  it("does not modify a file that already starts with the same banner", async () => {
    const root = await setup();
    const file = join(root, "a.ts");
    const original = "/* banner */\nconst value = 1;\n";
    await writeFile(file, original);

    await expect(prependBanner(file, "/* banner */\n\n")).resolves.toBe(false);
    await expect(readFile(file, "utf8")).resolves.toBe(original);
  });

  it("writes only the banner when the original file is empty", async () => {
    const root = await setup();
    const file = join(root, "empty.ts");
    await writeFile(file, "");

    await expect(prependBanner(file, "/* banner */\n")).resolves.toBe(true);
    await expect(readFile(file, "utf8")).resolves.toBe("/* banner */");
  });

  it("throws when the target file does not exist", async () => {
    const root = await setup();

    await expect(prependBanner(join(root, "missing.ts"), "/* banner */"))
      .rejects.toThrow();
  });
});

import { afterEach, describe, expect, it, vi } from "vitest";

import { headify } from "../src/headify.js";

describe("headify", () => {
  let project: Awaited<ReturnType<typeof createProject>> | undefined;

  afterEach(async () => {
    vi.restoreAllMocks();
    await project?.cleanup();
    project = undefined;
  });

  it("returns an empty result and skips package.json loading when no files match", async () => {
    project = await createProject();

    const result = await headify(["missing/**/*.ts"], { cwd: project.cwd });

    expect(result).toMatchObject({
      template: "",
      dryRun: false,
      total: 0,
      updated: 0,
      unchanged: 0,
      wouldUpdate: 0,
      files: [],
    });
  });

  it("prepends a banner to matching files and returns summary counts", async () => {
    project = await createProject();
    await project.writeFile("src/index.ts", "export const value = 1;\n");
    await project.writeFile("src/other.ts", "export const other = 2;\n");

    const result = await headify(["src/**/*.ts"], {
      cwd: project.cwd,
      template: "",
    });

    expect(result.total).toBe(2);
    expect(result.updated).toBe(2);
    expect(result.unchanged).toBe(0);
    expect(result.wouldUpdate).toBe(0);
    expect(result.files).toEqual(
      expect.arrayContaining([
        { path: "src/index.ts", status: "updated" },
        { path: "src/other.ts", status: "updated" },
      ]),
    );
    expect(await project.readFile("src/index.ts")).toContain("export const value = 1;");
    expect(await project.readFile("src/index.ts")).toMatch(/^\/\*!/);
  });

  it("does not add a duplicate banner when run more than once", async () => {
    project = await createProject();
    await project.writeFile("src/index.ts", "export const value = 1;\n");

    const options = { cwd: project.cwd, template: "" };
    const first = await headify(["src/index.ts"], options);
    const contentAfterFirstRun = await project.readFile("src/index.ts");
    const second = await headify(["src/index.ts"], options);

    expect(first.updated).toBe(1);
    expect(second.updated).toBe(0);
    expect(second.unchanged).toBe(1);
    expect(await project.readFile("src/index.ts")).toBe(contentAfterFirstRun);
  });

  it("excludes files matching exclude patterns", async () => {
    project = await createProject();
    await project.writeFile("src/index.ts", "index\n");
    await project.writeFile("dist/bundle.js", "bundle\n");

    const result = await headify(["**/*"], {
      cwd: project.cwd,
      exclude: ["dist/**", "package.json"],
      template: "",
    });

    expect(result.files.map((file) => file.path)).toEqual(["src/index.ts"]);
    expect(await project.readFile("dist/bundle.js")).toBe("bundle\n");
  });

  it("dry-run reports files that would change without writing", async () => {
    project = await createProject();
    const original = "export const value = 1;\n";
    await project.writeFile("src/index.ts", original);

    const result = await headify(["src/index.ts"], {
      cwd: project.cwd,
      dryRun: true,
      template: "",
    });

    expect(result.dryRun).toBe(true);
    expect(result.updated).toBe(0);
    expect(result.wouldUpdate).toBe(1);
    expect(result.unchanged).toBe(0);
    expect(result.files).toEqual([{ path: "src/index.ts", status: "wouldUpdate" }]);
    expect(await project.readFile("src/index.ts")).toBe(original);
  });

  it("dry-run reports an existing banner as unchanged", async () => {
    project = await createProject();
    await project.writeFile("src/index.ts", "/* custom */\nsource\n");
    await project.writeFile("template.mjs", "export default () => '/* custom */';");

    const result = await headify(["src/index.ts"], {
      cwd: project.cwd,
      template: join(project.cwd, "template.mjs"),
      dryRun: true,
    });

    expect(result.wouldUpdate).toBe(0);
    expect(result.unchanged).toBe(1);
  });

  // it("uses the custom template and provides file/package context", async () => {

  //   await project.writeFile(

  //     "export default (ctx) => `// ${ctx.package.name}@${ctx.package.version} | ${ctx.file.path} | ${ctx.file.basename}${ctx.file.ext}`;",
  //   );

  //   const result = await headify(["src/index.ts"], {
  //     cwd: project.cwd,
  //     template: join(project.cwd, "banner.mjs"),
  //   });

  //   expect(result.template).toBe("banner.mjs");
  //   expect(await project.readFile("src/index.ts"))
  //     .toMatch(/^\/\/ test-project@1\.2\.3 \| src\/index\.ts \| index\.ts\nsource\n$/);
  // });

  it("validates patterns and option types", async () => {
    project = await createProject();

    await expect(headify("src/*.ts" as never, { cwd: project.cwd })).rejects.toThrow(
      "Expected patterns to be an array of strings.",
    );
    await expect(headify([], { cwd: 123 as never })).rejects.toThrow(
      'Expected "cwd" to be a string.',
    );
    await expect(headify([], { cwd: project.cwd, template: 123 as never })).rejects.toThrow(

    await expect(headify([], { cwd: project.cwd, exclude: "dist/**" as never })).rejects.toThrow(

test/helpers.ts

import { mkdtemp, readFile, rm, writeFile, mkdir } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join, dirname } from "node:path";

export async function createProject() {
  const cwd = await mkdtemp(join(tmpdir(), "headmark-test-"));

  await writeFile(
    join(cwd, "package.json"),
    JSON.stringify({
      name: "test-project",
      version: "1.2.3",
      homepage: "https://example.com",
      author: "Test Author",
      license: "MIT",
    }, null, 2),
  );

  return {
    cwd,
    async writeFile(relativePath: string, content: string) {
      const path = join(cwd, relativePath);
      await mkdir(dirname(path), { recursive: true });
      await writeFile(path, content, "utf8");
      return path;
    },
    readFile(relativePath: string) {
      return readFile(join(cwd, relativePath), "utf8");
    },
    async cleanup() {
      await rm(cwd, { recursive: true, force: true });
    },
  };
}


test/renderer.test.ts

import { mkdtemp, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { afterEach, describe, expect, it, vi } from "vitest";
import { renderBanner, resolveTemplate } from "../src/renderer.js";
import loadTemplate from "../src/load-template.js";

let cwd: string;

afterEach(async () => {
  vi.restoreAllMocks();
  if (cwd) await rm(cwd, { recursive: true, force: true });
});

describe("resolveTemplate", () => {
  it("returns the default template when no path is supplied", async () => {
    const template = await resolveTemplate();

    expect(template).toBeTypeOf("function");
  });

  it("loads an ESM JavaScript template", async () => {
    cwd = await mkdtemp(join(tmpdir(), "headmark-template-"));
    const path = join(cwd, "template.mjs");
    await writeFile(path, "export default (ctx) => `// ${ctx.file.name}`;", "utf8");

    const template = await resolveTemplate(path);

    await expect(Promise.resolve(template({
      package: { name: "demo" },
      year: 2026,
      file: { path: "src/a.ts", name: "a.ts", basename: "a", ext: ".ts" },
    }))).resolves.toBe("// a.ts");
  });

  it("loads a CommonJS template", async () => {
    cwd = await mkdtemp(join(tmpdir(), "headmark-template-"));
    const path = join(cwd, "template.cjs");
    await writeFile(path, "module.exports = (ctx) => '// ' + ctx.file.name;", "utf8");

    const template = await loadTemplate(path);

    expect(await template({
      package: { name: "demo" },
      year: 2026,
      file: { path: "a.js", name: "a.js", basename: "a", ext: ".js" },
    })).toBe("// a.js");
  });

  it("rejects unsupported template extensions", async () => {
    cwd = await mkdtemp(join(tmpdir(), "headmark-template-"));
    const path = join(cwd, "template.json");
    await writeFile(path, "{}", "utf8");

    await expect(loadTemplate(path)).rejects.toThrow("Unsupported template extension: .json");
  });

  it("rejects templates that do not export a function", async () => {
    cwd = await mkdtemp(join(tmpdir(), "headmark-template-"));
    const path = join(cwd, "template.mjs");
    await writeFile(path, "export default 'not a function';", "utf8");

    await expect(loadTemplate(path)).rejects.toThrow("must export a function");
  });
});

describe("renderBanner", () => {
  const context = {
    package: { name: "demo" },
    year: 2026,
    file: { path: "src/a.ts", name: "a.ts", basename: "a", ext: ".ts" },
  };

  it("returns a synchronous template result", async () => {
    await expect(renderBanner(() => "/* banner */", context)).resolves.toBe("/* banner */");
  });

  it("supports asynchronous template functions", async () => {
    await expect(renderBanner(async () => "/* async banner */", context))
      .resolves.toBe("/* async banner */");
  });

  it("throws if the template result is not a string", async () => {
    await expect(renderBanner(() => 123 as never, context))
      .rejects.toThrow("Failed to render banner template.");
  });
});

import { afterEach, describe, expect, it, vi } from "vitest";

import { logger } from "../src/logger.js";
import { reportResult } from "../src/reporter.js";

describe("reportResult", () => {
  afterEach(() => {
    vi.restoreAllMocks();
  });

  it("warns and returns when no files matched", () => {
    const warn = vi.spyOn(logger, "warn").mockImplementation(() => {});
    const info = vi.spyOn(logger, "info").mockImplementation(() => {});
    const success = vi.spyOn(logger, "success").mockImplementation(() => {});

    reportResult("1.0.0", {
      template: "",
      dryRun: false,
      total: 0,
      updated: 0,
      unchanged: 0,
      wouldUpdate: 0,
      elapsed: 1,
      files: [],
    });

    expect(warn).toHaveBeenCalledWith("No files matched the specified patterns.");
    expect(info).not.toHaveBeenCalled();
    expect(success).not.toHaveBeenCalled();
  });

  it("reports normal run totals and file statuses", () => {
    const info = vi.spyOn(logger, "info").mockImplementation(() => {});
    const success = vi.spyOn(logger, "success").mockImplementation(() => {});

    reportResult("1.2.3", {
      template: "",
      dryRun: false,
      total: 2,
      updated: 1,
      unchanged: 1,
      wouldUpdate: 0,
      elapsed: 12,
      files: [
        { path: "src/index.ts", status: "updated" },
        { path: "src/other.ts", status: "unchanged" },
      ],
    });

    expect(info).toHaveBeenCalledWith("headify v1.2.3");
    expect(info).toHaveBeenCalledWith("Matched 2 files");
    expect(info).toHaveBeenCalledWith("Template: default");
    expect(info).toHaveBeenCalledWith("Updated: 1 Unchanged: 1");
    expect(info).toHaveBeenCalledWith(expect.stringContaining("src/index.ts"));
    expect(info).toHaveBeenCalledWith(expect.stringContaining("src/other.ts"));
    expect(success).toHaveBeenCalledWith(expect.stringContaining("Completed in"));
  });

  it("reports dry-run mode and would-update totals", () => {
    const info = vi.spyOn(logger, "info").mockImplementation(() => {});

    reportResult("1.2.3", {
      template: "banner.mjs",
      dryRun: true,
      total: 1,
      updated: 0,
      unchanged: 0,
      wouldUpdate: 1,
      elapsed: 4,
      files: [{ path: "src/index.ts", status: "wouldUpdate" }],
    });

    expect(info).toHaveBeenCalledWith("Template: banner.mjs");
    expect(info).toHaveBeenCalledWith("Dry run: no files will be modified");
    expect(info).toHaveBeenCalledWith("Would update: 1 Unchanged: 0");
  });
});
